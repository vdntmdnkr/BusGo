A. CREATE — Register User (src/controllers/authController.js)

const register = async (req, res, next) => {
  try {
    const { name, email, password } = req.body;
    const existing = await User.findOne({ email });
    if (existing) return res.status(400).json({ success: false, message: 'Email already registered.' });

    const user = await User.create({ name, email, password }); // password auto-hashed
    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '7d' });
    res.status(201).json({ success: true, token, user: user.toPublicJSON() });
  } catch (err) { next(err); }
};

// Frontend UI call (script.js)

  method: 'POST',

B. READ — Search Buses (src/controllers/busController.js)

const searchBuses = async (req, res, next) => {
  try {
    const { from, to } = req.query;
    const buses = await Bus.find({
      from: from.toLowerCase(), to: to.toLowerCase(),
      isActive: true, availableSeats: { $gt: 0 }
    }).sort({ fare: 1 });
    res.status(200).json({ success: true, count: buses.length, data: buses });
  } catch (err) { next(err); }
};

// Frontend UI call (script.js)

C. UPDATE — Edit Profile (src/controllers/authController.js)

const updateMe = async (req, res, next) => {
  try {
    const updates = {};
    ['name', 'phone', 'city', 'gender'].forEach(f => {
      if (req.body[f] !== undefined) updates[f] = req.body[f];
    });
    const updated = await User.findByIdAndUpdate(req.user._id, { $set: updates }, { new: true });
    res.status(200).json({ success: true, message: 'Profile updated.', user: updated.toPublicJSON() });
  } catch (err) { next(err); }
};

// Frontend UI call (script.js)
const res = await apiFetch('/auth/me', {
  method: 'PUT',
  body: JSON.stringify({ name, phone, city })
});


D. DELETE — Cancel Booking (src/controllers/bookingController.js)

const cancelBooking = async (req, res, next) => {
  try {
    const booking = await Booking.findById(req.params.id);
    if (!booking) return res.status(404).json({ success: false, message: 'Booking not found.' });

    booking.status = 'cancelled';     // Soft delete — record kept for audit
    booking.paymentStatus = 'refunded';
    await booking.save();

    await User.findByIdAndUpdate(req.user._id, { $inc: { walletBalance: booking.totalFare } });
    res.status(200).json({ success: true, message: 'Booking cancelled. Refund credited to wallet.' });
  } catch (err) { next(err); }
};

// Frontend UI call (script.js)
const res = await apiFetch(`/bookings/${bookingId}/cancel`, { method: 'PUT' });
